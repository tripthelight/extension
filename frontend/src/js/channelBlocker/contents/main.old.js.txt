/**
 * @param {Array<string>} channelNames 화면에 보이는 모든 Shorts 썸네일의 video-id 모음 배열
 * @param {Array<{channelName: string, videoId: string}>} requestData 화면에 보이는 모든 Shorts 썸네일의 video-id 를 googleapi에 요청 후 응답받은 data
 */
function addBlockingChannelClass(channelNames, requestData) {
  // 화면에 Shorts 영상이 추가된 경우 재조회된 data를 DATAS.views.shorts.data에 교체
  const isSame = DATAS.views.shorts.data.length === requestData.length;
  if (isSame) {
    //
  } else {
    DATAS.views.shorts.data.length = 0;
    DATAS.views.shorts.data.push(...requestData);
  }

  const blockedVideoIds = requestData
    .filter((item) => channelNames.includes(item.channelName))
    .map((item) => item.videoId);

  for (const selector of SHORTS_TAG.contents) {
    const elements = document.querySelectorAll(selector);

    elements.forEach((element) => {
      const aTags = element.querySelectorAll("a[href]");
      for (const aTag of aTags) {
        const href = aTag.getAttribute("href") || "";
        const matched = blockedVideoIds.some((videoId) => href.includes(videoId));
        if (matched) {
          element.classList.add("blocking-channel");
          break;
        }
      }
    });
  }
}

function escapeRegExp(str) {
  return str.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}
/**
 * 화면에 있는 모든 Shorts 영상의 video-id를 모아서 리턴
 * 여기에 해당하는 Shorts 썸네일에는 video-id 정보만 존재함
 * 그래서 video-id를 모아서 googleapi에 요청 보내야 함
 * @returns {Array<string>} Shorts video-id 모음 배열
 */
function getShortsVideoIds() {
  const videoIds = [];
  const seen = new Set();

  const shortsPath = DATAS.views.shorts.path; // 원본 그대로 사용
  const shortsPathForRegex = shortsPath.endsWith("/")
    ? shortsPath.slice(0, -1)
    : shortsPath;

  const shortsRegex = new RegExp(
    `${escapeRegExp(shortsPathForRegex)}/([^/?#]+)`
  );

  for (const selector of SHORTS_TAG.contents) {
    const elements = document.querySelectorAll(selector);

    elements.forEach((element) => {
      // ———————————————————————————————
      // element 조회 시 코드에만 있고, 화면에는 없는 중복 tag 조회 방지
      // ———————————————————————————————
      if (location.pathname === "/" && !element.closest(`${DATAS.views.main.wrap}`)) return; // index
      else if (location.pathname === "/results" && !element.closest(`${DATAS.views.results.wrap}`)) return; // 검색 후 화면
      else if (location.pathname === "/watch" && !element.closest(`${DATAS.views.watch.wrap}`)) return; // 상세 화면
      else if ((location.pathname.startsWith('/@') || /^\/@/.test(location.pathname)) || (location.pathname.startsWith(`${DATAS.views.channel.path}`) || /^\/@/.test(`${DATAS.views.channel.path}`)) && !element.closest(`${DATAS.views.channel.wrap}`)) return; // 채널 페이지
      else if (location.pathname.startsWith(`${DATAS.views.shorts.path}`) || /^\/@/.test(`${DATAS.views.shorts.path}`) && !element.closest(`${DATAS.views.shorts.wrap}`)) return; // 쇼츠화면
      // ———————————————————————————————
      
      const aTags = element.querySelectorAll("a[href]");

      aTags.forEach((aTag) => {
        const href = aTag.getAttribute("href") || "";

        if (href.startsWith(shortsPathForRegex + "/") || href.includes(shortsPathForRegex + "/")) {
          const match = href.match(shortsRegex);

          if (match && match[1]) {
            const videoId = match[1];

            if (!seen.has(videoId)) {
              seen.add(videoId);
              videoIds.push(videoId);
            }
          }
        }
      });
    });
  };

  // TODO: 이상하게 꼬임
  // google youtube api는 video-id를 50개까지만 받음
  // 그래서 새로 추가된 video-id만 적용
  // const oldIdSet = new Set(DATAS.views.shorts.oldIds);
  // const remainderIds = videoIds.filter((id) => !oldIdSet.has(id));

  // DATAS.views.shorts.oldIds.length = 0;
  // DATAS.views.shorts.oldIds.push(...videoIds);

  // return remainderIds;
  
  return videoIds;
}

/**
 * 요소가 화면에 얼마나 보이는지 계산
 * 가장 많이 보이는 요소를 "현재 Shorts"로 간주
 */
function getVisibleRatio(el) {
  if (!el || !el.isConnected) return 0;
  const rect = el.getBoundingClientRect();
  const vw = window.innerWidth;
  const vh = window.innerHeight;

  const visibleWidth = Math.max(0, Math.min(rect.right, vw) - Math.max(rect.left, 0));
  const visibleHeight = Math.max(0, Math.min(rect.bottom, vh) - Math.max(rect.top, 0));

  const visibleArea = visibleWidth * visibleHeight;
  const totalArea = Math.max(1, rect.width * rect.height);

  return visibleArea / totalArea;
};
/**
 * 현재 화면에서 가장 많이 보이는 Shorts 요소 찾기
 */
function getCurrentShortsElement() {
  const items = [...document.querySelectorAll(SHORTS_TAG.list)];
  if (!items.length) return null;

  let bestEl = null;
  let bestRatio = -1;

  for (const el of items) {
    const ratio = getVisibleRatio(el);
    if (ratio > bestRatio) {
      bestRatio = ratio;
      bestEl = el;
    }
  }

  return bestEl;
};
/**
 * 현재 DOM 순서 기준으로 다음 요소 찾기
 */
function getNextShortsElement(currentEl) {
  if (!currentEl) return null;

  let next = currentEl.nextElementSibling;
  while (next) {
    if (next.matches?.(SHORTS_TAG.list)) return next;
    next = next.nextElementSibling;
  }

  return null;
};
/**
 * 현재 DOM 순서 기준으로 이전 요소 찾기
 */
function getPrevShortsElement(currentEl) {
  if (!currentEl) return null;

  let prev = currentEl.previousElementSibling;
  while (prev) {
    if (prev.matches?.(SHORTS_TAG.list)) return prev;
    prev = prev.previousElementSibling;
  }

  return null;
};
/**
 * 대상 요소를 화면 중앙으로 가져오기
 * 유튜브 Shorts의 내부 스크롤/슬라이드 상태를 최대한 존중하기 위해
 * scrollIntoView를 사용
 */
function moveToShorts(targetEl) {
  if (!targetEl || !targetEl.isConnected) return false;

  targetEl.scrollIntoView({
    block: 'center',
    inline: 'nearest',
    behavior: 'auto',
  });

  return true;
};
/**
 * 현재 보고 있는 Shorts를 제거하고 다음 Shorts로 이동
 *
 * 동작 순서:
 * 1) 현재 요소 찾기
 * 2) 다음 요소 찾기 (없으면 이전 요소)
 * 3) 먼저 대상 요소로 이동
 * 4) 잠깐 기다린 뒤 현재 요소 제거
 */
async function removeCurrentShortsAndSlide() {
  const currentEl = getCurrentShortsElement();
  if (!currentEl) {
    console.warn('[ShortsRemover] 현재 Shorts 요소를 찾지 못했습니다.');
    return false;
  }

  const nextEl = getNextShortsElement(currentEl);
  const prevEl = getPrevShortsElement(currentEl);

  // 우선순위: 다음 영상 -> 이전 영상
  const targetEl = nextEl || prevEl;

  // 삭제할 대상의 id는 언제든 바뀔 수 있으므로,
  // id 숫자에 의존하지 않고 DOM 위치 기준으로 처리
  if (!targetEl) {
    console.warn('[ShortsRemover] 이동할 다음/이전 Shorts가 없습니다. 현재 요소만 제거합니다.');
    currentEl.remove();
    return true;
  }

  // 1차 이동
  moveToShorts(targetEl);

  // 브라우저 렌더링/유튜브 내부 상태 반영 대기
  await new Promise((resolve) => requestAnimationFrame(resolve));
  await new Promise((resolve) => requestAnimationFrame(resolve));

  // 혹시 한 번 더 보정
  moveToShorts(targetEl);

  // 내부 상태가 target으로 넘어갈 시간을 조금 더 줌
  await new Promise((resolve) => setTimeout(resolve, 80));

  // 현재 요소 제거
  if (currentEl.isConnected) {
    currentEl.remove();
  }

  // 제거 후 target으로 다시 한번 고정
  await new Promise((resolve) => requestAnimationFrame(resolve));
  moveToShorts(targetEl);

  return true;
};
/**
 * 원하는 특정 Shorts 요소를 제거하고 다음으로 넘기고 싶을 때 사용하는 함수
 * 예: id가 "3"인 요소를 직접 넘겨서 제거
 *
 * 단, 실제 사용에서는 "현재 화면에 보이는 요소" 제거가 더 안전합니다.
 */
async function removeSpecificShortsAndSlide(el) {
  if (!el || !el.isConnected || !el.matches?.(ITEM_SELECTOR)) {
    console.warn('[ShortsRemover] 유효한 Shorts 요소가 아닙니다.');
    return false;
  }

  const nextEl = getNextShortsElement(el);
  const prevEl = getPrevShortsElement(el);
  const targetEl = nextEl || prevEl;

  if (!targetEl) {
    el.remove();
    return true;
  }

  moveToShorts(targetEl);

  await new Promise((resolve) => requestAnimationFrame(resolve));
  await new Promise((resolve) => requestAnimationFrame(resolve));

  moveToShorts(targetEl);

  await new Promise((resolve) => setTimeout(resolve, 80));

  if (el.isConnected) {
    el.remove();
  }

  await new Promise((resolve) => requestAnimationFrame(resolve));
  moveToShorts(targetEl);

  return true;
};

// 채널 추천 안함 채널 제거 - Shorts 페이지용
function removeBlockerShortsPage() {
  const isShortsPage = location.pathname.startsWith(`${DATAS.views.shorts.path}`) || /^\/@/.test(`${DATAS.views.shorts.path}`);
  if (!isShortsPage) return;

  const videoId = location.pathname.match(/^\/shorts\/([^/?#]+)/)?.[1];
  responseChannelName(videoId)
    .then((channelData) => {
      if (!channelData) return;
      const { channelName, channelUrl } = channelData;
      if (!channelName || channelName === "") return;
      if (!channelUrl || channelUrl === "") return;

      // long form 제거
      // Read DataBase
      if (!db) return;
      const store = db.transaction("b", "readonly").objectStore("b");
      const getBlobReq = store.get("channelNames");
      getBlobReq.addEventListener("success", async (event) => {
        // 채널 추천 안함 채널명은 indexedDB에서 노출방지를 위한 blob 객체로 관리
        const blob = event.target.result;
        if (!blob) return; 
        const text = await blob.text();
        const channelNames = JSON.parse(text);

        if (!channelNames.includes(channelName)) return;

        const blockShortsUrl = decodeURIComponent(channelUrl.replace(/^.*(@[^/?#]+).*$/, "$1"));
        if (!blockShortsUrl || blockShortsUrl === "") return;

        // 차단된 채널일 경우 
        // Shorts 제거 후,
        // 전체 다음 버튼 클릭
        // 채널 안에 다른 채널명을 가진 차단 채널 썸네일이 있을 경우 remove
        document.querySelectorAll(SHORTS_TAG.list).forEach((reel) => {
          reel.querySelectorAll("a").forEach(async (a) => {
            if (a.textContent.trim() !== blockShortsUrl) return;
            await ShortsRemover.removeCurrentShortsAndSlide();
          });
        });
      });
      getBlobReq.addEventListener("error", (event) => {
        const error = event.target.error;
        throw new Error(`db readAll failed: ${error.name}`);
      });

    })
    .catch((err) => {
      throw new Error(err);
    });
}

// 채널 추천 안함 채널 제거 - 채널 페이지용
function removeBlockerThumbChannelPage() {
  // 채널 페이지 여부 체크 - location.pathname이 "/@" or "/channel/"로 시작하면 true
  const isChannelPage = (location.pathname.startsWith("/@") || location.pathname.startsWith("/channel/")) || (/^\/(@|channel\/)/.test(location.pathname));
  if (!isChannelPage) return;

  // 차단된 채널일 경우 전체 content 제거
  // Read DataBase
  if (!db) return;
  const store = db.transaction("b", "readonly").objectStore("b");
  const getBlobReq = store.get("channelNames");
  getBlobReq.addEventListener("success", async (event) => {
    // 채널 추천 안함 채널명은 indexedDB에서 노출방지를 위한 blob 객체로 관리
    const blob = event.target.result;
    if (!blob) return; 
    const text = await blob.text();
    const channelNames = JSON.parse(text);

    // 채널 안에 다른 채널명을 가진 차단 채널 썸네일이 있을 경우 hide
    const listSelector = LIST_TAG.join(",");
    document.querySelectorAll(listSelector).forEach((item) => {
      const text = item.textContent?.trim();
      if (!text) return;
      const hasChannelName = channelNames.some((channelName) =>
        text.includes(channelName)
      );
      if (hasChannelName) {
        item.classList.add("blocking-recomn");
      }
    });

    // 차단 채널 전체 hide
    const chNameElements = document.querySelectorAll(CHANNEL_TAG.chName);
    const contentsElements = document.querySelectorAll(CHANNEL_TAG.contents);
    const hasMatchedChannelName = [...chNameElements].some((chNameEl) => {
      const targets = [chNameEl, ...chNameEl.querySelectorAll("*")];
      return targets.some((el) =>
        channelNames.some((channelName) =>
          el.textContent?.includes(channelName)
        )
      );
    });
    if (!hasMatchedChannelName) return;
    contentsElements.forEach((el) => {
      el.classList.add("blocking-channel");
    });
  });
  getBlobReq.addEventListener("error", (event) => {
    const error = event.target.error;
    throw new Error(`db readAll failed: ${error.name}`);
  });
};

// 채널명 조회
function findChannelName(_id, _ids, _case) {
  if (_case === "blocker") { // 채널 추천 안함
    // _id : channelName
    // _ids : channelNames
    console.log("채널 추천 안함 채널명 : ", _id);

    const channelNamesSet = [...new Set(_ids.concat(_id))];
    // 채널 추천 안함 채널명은 indexedDB에서 노출방지를 위한 blob 객체로 관리
    const blob = new Blob(
      [JSON.stringify(channelNamesSet)],
      { type: "application/json" }
    );

    // Write DataBase
    const store = db.transaction("b", "readwrite").objectStore("b");
    const addReq = store.put(blob, "channelNames");
    addReq.addEventListener("success", (event) => {
      // console.log("indexedDB write Data result : ", event.target.result);
      // 현재 화면에서 channelNames 를 가진 썸네일 제거
      removeBlockerThumb();
    });
    addReq.addEventListener("error", (event) => {
      const error = event.target.error;
      throw new Error(`db write failed: ${error.name}`);
    });

  } else if (_case === "interest") { // 관심 없음
    // _id : videoId
    // _ids : videoIds
    console.log("관심 없는 video Id : ", _id);

    const videoIdsSet = [...new Set(_ids.concat(_id))];
    // 관심없는 채널명은 indexedDB에서 노출방지를 위한 blob 객체로 관리
    const blob = new Blob(
      [JSON.stringify(videoIdsSet)],
      { type: "application/json" }
    );

    // Write DataBase
    const store = db.transaction("i", "readwrite").objectStore("i");
    const addReq = store.put(blob, "videoIds");
    addReq.addEventListener("success", (event) => {
      // console.log("indexedDB write Data result : ", event.target.result);
      
      // 현재 화면에서 videoId 를 가진 썸네일 제거
      removeInterestThumb();
    });
    addReq.addEventListener("error", (event) => {
      const error = event.target.error;
      throw new Error(`db write failed: ${error.name}`);
    });
  }
};

function extractVideoId(href) {
  if (typeof href !== 'string' || !href.trim()) return null;

  return (
    href.match(/[?&]v=([^&]+)/)?.[1] ??
    href.match(/\/shorts\/([^?&/]+)/)?.[1] ??
    null
  );
};

function collectShortsVideoIds() {
  return [...new Set(
    [...document.querySelectorAll('a[href*="/shorts/"]')]
      .map((a) => extractVideoId(a.href))
      .filter(Boolean)
  )];
};

function getVodData(href) {
  const VOD_DATA = {
    videoId: null,
    videoIds: null,
  };

  const videoId = extractVideoId(href);

  if (!videoId) {
    return JSON.stringify(VOD_DATA);
  }

  VOD_DATA.videoId = videoId;
  VOD_DATA.videoIds = collectShortsVideoIds();

  return JSON.stringify(VOD_DATA);
};

  responseChannelName(videoId)

      // Read DataBase
      if (!db) return;
      const store = db.transaction("b", "readonly").objectStore("b");
      const getBlobReq = store.get("channelNames");
      getBlobReq.addEventListener("success", async (event) => {

        const error = event.target.error;
        throw new Error(`db readAll failed: ${error.name}`);
      });
    })
    .catch((err) => {
      throw new Error(err);
    });

  dummyElementClick();
};

function ensureMenuButtons(listElement, videoObj) {
  if (!listElement) return;

  let btnBlocker = listElement.querySelector(`.${DATAS.btns.blocker.cls}`);
  let btnInterest = listElement.querySelector(`.${DATAS.btns.interest.cls}`);

  if (!btnBlocker) {
    btnBlocker = createMenuButton({
      className: DATAS.btns.blocker.cls,
      text: DATAS.btns.blocker.text,
      onClick: btnBlockerEvent,
    });
    listElement.appendChild(btnBlocker);
  }

  if (!btnInterest) {
    btnInterest = createMenuButton({
      className: DATAS.btns.interest.cls,
      text: DATAS.btns.interest.text,
      onClick: btnInterestEvent,
    });
    listElement.appendChild(btnInterest);
  }

  delete btnBlocker.dataset.videoId;
  delete btnBlocker.dataset.videoIds;
  delete btnInterest.dataset.videoId;
  delete btnInterest.dataset.videoIds;

  if (getParamType(videoObj) === "object") {
    const { videoIds = [], videoId = "" } = JSON.parse(videoObj);
    btnBlocker.dataset.videoIds = JSON.stringify(videoIds);;
    btnBlocker.dataset.videoId = videoId;
    btnInterest.dataset.videoIds = JSON.stringify(videoIds);
    btnInterest.dataset.videoId = videoId;
  } else if (getParamType(videoObj) === "string") {
    btnBlocker.dataset.videoId = videoObj;
    btnInterest.dataset.videoId = videoObj;
  }
};

// [ 메인화면, 검색 후 화면, 상세화면, 채널페이지, 재생목록 ] 구분
function getClickedVideoId(target, path) {
  // 재생목록
  if (target?.closest(DATAS.views.playlist.lists)) {
    return findVideoIdFromAnchors(target?.closest(DATAS.views.playlist.list));
  }

  // 메인화면 - Shorts 리스트 있음
  if (path === DATAS.views.main.path) {
    return findVideoIdFromAnchors(target?.closest(DATAS.views.main.list));
  }

  // 검색 후 화면 - Shorts 리스트 있음
  if (path === DATAS.views.results.path) {
    return findVideoIdFromAnchors(
      findClosestBySelectors(target, DATAS.views.results.list)
    );
  }

  // 상세화면 - Shorts 리스트 있음
  if (path === DATAS.views.watch.path) {
    if (target?.closest(DATAS.views.watch.current)) {
      return new URL(location.href).searchParams.get("v");
    }

    return findVideoIdFromAnchors(
      findClosestBySelectors(target, DATAS.views.watch.recommends)
    );
  }

  // 채널 페이지 - Shorts 리스트 있음
  if (
    // location.pathname 이 "/@" 로 시작
    (
      location.pathname.startsWith('/@') ||
      /^\/@/.test(location.pathname)
    ) || 
    // location.pathname 이 "/channel" 로 시작
    (
      location.pathname.startsWith(`${DATAS.views.channel.path}`) ||
      /^\/@/.test(`${DATAS.views.channel.path}`)
    )
  ) {
    return findVideoIdFromAnchors(
      findClosestBySelectors(target, DATAS.views.channel.list)
    );
  }

  // Shorts 화면
  if (location.pathname.startsWith(`${DATAS.views.shorts.path}`) || /^\/@/.test(`${DATAS.views.shorts.path}`)) {
    return location.pathname.match(/^\/shorts\/([^/?#]+)/)?.[1];
  }

  return null;
};

function clearActiveInterval() {
  if (STATE.activeIntervalId) {
    clearInterval(STATE.activeIntervalId);
    STATE.activeIntervalId = null;
  }
};

function pollVideoIdAndOpenMenu(target) {
  clearActiveInterval();

  let attempt = 0;

  STATE.activeIntervalId = setInterval(() => {
    attempt += 1;

    const id = getClickedVideoId(target, location.pathname);

    if (id) {
      clearActiveInterval();
      popupMenuHandler(id);
      return;
    }

    if (attempt >= MAX_RETRY) {
      clearActiveInterval();
    }
  }, RETRY_DELAY_MS);
};

function updateThumbObserve() {
  if (STATE.thumbObserver) return;
  const SELECTOR = LIST_TAG.join(",");

  STATE.thumbObserver = new MutationObserver((mutations) => {
    let shouldUpdate = false;

    mutations.forEach((mutation) => {
      mutation.addedNodes.forEach((node) => {
        if (node.nodeType !== 1) return;

        if (
          node.matches?.(SELECTOR) ||
          node.querySelector?.(SELECTOR)
        ) {
          shouldUpdate = true;
        }
      });
    });

    if (shouldUpdate) {
      scheduleRemoveVodThumb();
    }
  });

  STATE.thumbObserver.observe(document.body, {
    childList: true,
    subtree: true,
  });
};

function wrapHistoryMethod(type) {
  const original = history[type];

  history[type] = function (...args) {
    const result = original.apply(this, args);
    handleUrlChange();
    return result;
  };
};

function watchUrlChanges() {
  if (STATE.isUrlWatcherBound) return;
  STATE.isUrlWatcherBound = true;

  wrapHistoryMethod("pushState");
  wrapHistoryMethod("replaceState");

  window.addEventListener("popstate", handleUrlChange);
  document.addEventListener("yt-navigate-finish", handleUrlChange);
};
